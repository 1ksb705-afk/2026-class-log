#include <stdio.h>

int main()
{
    char blood;
    printf("혈액형 입력(A,B,O):");
    scanf(" %c", &blood);
    
    //조건은 switch ~case
    switch(blood)
    {
      case 'A':
        printf("A형\n");
        break;
      case 'B':
        printf("B형\n");
        break;
      case 'O':
        printf("O형\n");
        break;
    default:
        printf("입력 오류\n");
    }
    return 0;
}
