#include <stdio.h>

int main()
{
    int menu;
      
    printf("[1:치킨 2:피자 3:떡볶이 4:햄버거 5~7: 랜덤]");
    scanf("%d", &menu);    
    
    if (menu == 1) 
        printf("치킨입니다.");       
    
        
    else if (menu ==2 ) 
        printf("피자입니다.");
     
    
    else if (menu ==3 ) 
        printf("떡볶이입니다.");
    
    
    else if (menu == 4 ) 
        printf("햄버거입니다.");
    
    else if (menu >= 5 && menu<=7 ) // 조건이 2개 이상일 때 둘 다 만족할 때 전체 만족이 된다.
        printf("랜덤");
    
    else
        printf("입력오류");
    
 
    return 0;
}
